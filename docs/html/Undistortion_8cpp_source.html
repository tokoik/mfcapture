37 bool readMatrix(const picojson::object& object, const std::string& key,
38 cv::Mat& matrix, int rows, int columns)
39 {
40 // 必須ノードが存在し、その値が JSON 配列であることを確認する。
41 const auto item{ object.find(key) };
42 if (item == object.end() || !item->second.is<picojson::array>()) return false;
43
44 // 要素数が期待する行列サイズと異なるファイルは受け付けない。
45 const auto& values{ item->second.get<picojson::array>() };
46 if (values.size() != static_cast<size_t>(rows * columns)) return false;
47
48 // 丸括弧で行数・列数・型を渡し、目的の二次元行列を確保する。
49 // 波括弧では initializer_list 版が選ばれて 3 x 1 行列になるので注意する。
50 cv::Mat loaded(rows, columns, CV_64F);
51
52 // JSON は行優先の一次元配列なので、商を行、剰余を列として格納する。
53 for (size_t i = 0; i < values.size(); ++i)
54 {
55 if (!values[i].is<double>()) return false;
56 loaded.at<double>(static_cast<int>(i) / columns,
57 static_cast<int>(i) % columns) = values[i].get<double>();
58 }
59
60 // 全要素を検証できた場合だけ呼び出し元の行列を更新する。
61 matrix = std::move(loaded);
62 return true;
63 }
64}

68//
69bool Undistortion::load(const std::string& filename)
70{
71 // 指定された JSON ファイルを開く。
72 std::ifstream stream{ Utf8ToTChar(filename) };
73 if (!stream) return false;
74
75 // picojson で構文解析し、ルートが JSON オブジェクトであることを確認する。
76 picojson::value value;
77 stream >> value;
78 if (!stream || !value.is<picojson::object>()) return false;
79
80 // 読み込み途中の失敗で現在の有効な較正値を壊さないよう、一時行列へ格納する。
81 cv::Mat loadedCamera;
82 cv::Mat loadedDistortion;
83 const auto& object{ value.get<picojson::object>() };
84
85 // calib の保存形式に合わせ、3 x 3 のカメラ行列と 5 x 1 の歪み係数を要求する。
86 if (!readMatrix(object, "camera matrix", loadedCamera, 3, 3)
87 || !readMatrix(object, "distortion", loadedDistortion, 5, 1))
88 {
89 return false;
90 }
91
92 // 両方の行列が正常な場合だけ、新しい較正値へまとめて置き換える。
93 cameraMatrix = std::move(loadedCamera);
94 distortion = std::move(loadedDistortion);
95
96 // 較正ファイルに画像サイズが記録されていれば読み込む。
97 imageSize = cv::Size{ 0, 0 };
98 auto sizeIt{ object.find("size") };
99 if (sizeIt == object.end()) sizeIt = object.find("image size");
100 if (sizeIt != object.end() && sizeIt->second.is<picojson::array>())
101 {
102 const auto& arr{ sizeIt->second.get<picojson::array>() };
103 if (arr.size() == 2 && arr[0].is<double>() && arr[1].is<double>())
104 {
105 imageSize = cv::Size{ static_cast<int>(arr[0].get<double>()), static_cast<int>(arr[1].get<double>()) };
106 }
107 }
108
109 // 較正値が変わったため、古い値から作った OpenCV の補正マップを無効化する。
110 mapX.release();
111 mapY.release();
112 mapSize = {};
113 return true;
114}

124//
125// OpenCV を使って入力画像のレンズ歪みを補正する
126//
127void Undistortion::apply(const cv::Mat& source, cv::Mat& destination)
128{
129 // パラメータまたは画像が無い場合は、補正せず入力をそのまま渡す。
130 if (!ready() || source.empty())
131 {
132 source.copyTo(destination);
133 return;
134 }
135
136 // 補正座標表は画像サイズに依存するため、サイズが変わったときだけ再計算する。
137 // カメラ行列は入力画像のサイズに換算してから使う。
138 if (mapSize != source.size())
139 {
140 const cv::Mat camera{ getCameraMatrix(source.size()) };
141 cv::initUndistortRectifyMap(camera, distortion, cv::Mat{},
142 camera, source.size(), CV_32FC1, mapX, mapY);
143 mapSize = source.size();
144 }
145
146 // 各出力画素が参照すべき入力座標をマップから求め、線形補間して補正画像を作る。
147 cv::remap(source, destination, mapX, mapY, cv::INTER_LINEAR,
148 cv::BORDER_CONSTANT);
149}

155{
156 if (a.width <= 0 || a.height <= 0 || b.width <= 0 || b.height <= 0) return false;

162//
163// 入力画像のサイズが較正時の画像サイズと合っているか調べる
164//
165CalibrationSizeMatch Undistortion::matchSize(const cv::Size& size) const
166{
167 // どちらかのサイズが分からなければ判定できない。
168 if (imageSize.width <= 0 || imageSize.height <= 0 || size.width <= 0 || size.height <= 0)
169 {
170 return CalibrationSizeMatch::Unknown;
171 }
172
173 // 同じ解像度ならそのまま使える。
174 if (size == imageSize) return CalibrationSizeMatch::Exact;
175
176 // アスペクト比が同じなら、多くのカメラで同じ視野を拡大縮小したものなので換算して使える。
177 if (isSameAspect(size, imageSize)) return CalibrationSizeMatch::Scaled;
178
179 // アスペクト比が違うモードは視野の切り出し方が違うので、換算しても正しくない。
180 return CalibrationSizeMatch::AspectMismatch;
181}

183//
184// 入力画像のサイズに換算したカメラ行列を得る
185//
186cv::Mat Undistortion::getCameraMatrix(const cv::Size& size) const
187{
188 // 換算できないときや同じ解像度のときは較正値をそのまま使う。
189 const auto match{ matchSize(size) };
190 if (match == CalibrationSizeMatch::Unknown || match == CalibrationSizeMatch::Exact)
191 {
192 return cameraMatrix;
193 }
194
195 // 幅と高さの比で焦点距離を拡大縮小し、主点は画素の中心を基準にして換算する。
196 const double sx{ static_cast<double>(size.width) / imageSize.width };
197 const double sy{ static_cast<double>(size.height) / imageSize.height };
198 cv::Mat scaled{ cameraMatrix.clone() };
199 scaled.at<double>(0, 0) *= sx;
200 scaled.at<double>(0, 1) *= sx;
201 scaled.at<double>(0, 2) = (scaled.at<double>(0, 2) + 0.5) * sx - 0.5;
202 scaled.at<double>(1, 1) *= sy;
203 scaled.at<double>(1, 2) = (scaled.at<double>(1, 2) + 0.5) * sy - 0.5;
204 return scaled;
205}
