42//
43// ArUco Marker を検出して画像に描画する
44//
45void Aruco::detectMarkers(cv::Mat& image, float markerLength,
46 const cv::Mat& cameraMatrix, const cv::Mat& distCoeffs)
47{
48 // 入力画像が空なら何もしない
49 if (image.empty()) return;
50
51 // 4 チャンネル画像の場合は OpenCV の検出・描画処理のために一時的に 3 チャンネル (BGR) 画像を作成する
52 cv::Mat tempImage;
53 const auto isFourChannels{ image.channels() == 4 };
54 if (isFourChannels)
55 {
56 cv::cvtColor(image, tempImage, cv::COLOR_BGRA2BGR);
57 }
58 else
59 {
60 tempImage = image;
61 }
62
63 // ArUco Marker のコーナーを検出する
64 detector->detectMarkers(tempImage, corners, ids, rejected);
65
66 // コーナーが見つからなければ戻る
67 if (corners.empty()) return;
68
69 // カメラ内部パラメータ行列が有効 (3x3) であればマーカーの姿勢推定と座標軸描画を行う
70 const bool hasCameraMatrix{ cameraMatrix.rows == 3 && cameraMatrix.cols == 3 };
71 if (hasCameraMatrix)
72 {
73 // 各マーカーの 4 隅に対応する 3 次元空間上の局所座標系 (Z=0 平面、中心原点)
74 const float markerCenter{ markerLength * 0.5f };
75 std::vector<cv::Point3f> markerObjPoints;
76 markerObjPoints.push_back(cv::Point3f(-markerCenter, markerCenter, 0.0f));
77 markerObjPoints.push_back(cv::Point3f(markerCenter, markerCenter, 0.0f));
78 markerObjPoints.push_back(cv::Point3f(markerCenter, -markerCenter, 0.0f));
79 markerObjPoints.push_back(cv::Point3f(-markerCenter, -markerCenter, 0.0f));
80
81 // 個々のマーカーについて
82 for (size_t i = 0; i < corners.size(); ++i)
83 {
84 // マーカーのコーナー検出位置から 3 次元姿勢（回転ベクトル・平行移動ベクトル）を推定する
85 cv::Vec3d rvec, tvec;
86 cv::solvePnP(markerObjPoints, corners[i], cameraMatrix, distCoeffs,
87 rvec, tvec, false, cv::SOLVEPNP_IPPE_SQUARE);
88
89 // マーカーの位置に座標軸を描く
90 cv::drawFrameAxes(tempImage, cameraMatrix, distCoeffs, rvec, tvec, markerLength);
91 }
92 }
93 else
94 {
95 // カメラパラメータが指定されていなければ、ArUco Marker の場所に矩形枠と ID 番号を描き込む
96 cv::aruco::drawDetectedMarkers(tempImage, corners, ids);
97 }
98
99 // 4 チャンネル画像の場合は検出・描画結果を元の画像へ書き戻す
100 if (isFourChannels)
101 {
102 cv::cvtColor(tempImage, image, cv::COLOR_BGR2BGRA);
103 }
104}
